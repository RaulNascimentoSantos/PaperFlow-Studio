1.1 CORREÇÃO DO SSE (Server-Sent Events)
-----------------------------------------
Arquivo: paperflow-api/src/routes/documents.ts

Implementar SSE compatível com EventSource nativo:
- Suporte a autenticação via query param ?apiKey= e header x-api-key
- Heartbeat a cada 15 segundos
- Suporte a reconexão com Last-Event-ID
- Formato correto: id:, event:, data:, retry:

Código completo:
```typescript
// No topo do arquivo
import { AuthService } from '@/services/auth';
import { createError } from '@/utils/errors';

fastify.get(
  '/:id/events',
  {
    schema: {
      description: 'Server-Sent Events stream for document processing',
      tags: ['Documents'],
      params: z.object({ id: z.string() }),
      querystring: z.object({ apiKey: z.string().optional() })
    },
  },
  async (request, reply) => {
    const { id } = request.params;
    const { apiKey } = request.query;
    const headerKey = request.headers['x-api-key'] as string | undefined;
    const key = headerKey ?? apiKey;

    if (!key) throw createError.unauthorized('Missing API key');
    const user = await AuthService.validateApiKey(key);
    if (!user) throw createError.unauthorized('Invalid API key');

    reply.raw.setHeader('Content-Type', 'text/event-stream');
    reply.raw.setHeader('Cache-Control', 'no-cache');
    reply.raw.setHeader('Connection', 'keep-alive');
    reply.raw.write('retry: 3000\n\n');

    let eventId = Number(request.headers['last-event-id'] ?? 0) || 0;

    const heartbeat = setInterval(() => {
      reply.raw.write(`:heartbeat ${Date.now()}\n\n`);
    }, 15000);

    const stages = [
      { stage: 'queued', progress: 0, message: 'Documento na fila' },
      { stage: 'parsing', progress: 20, message: 'Analisando PDF' },
      { stage: 'extracting', progress: 40, message: 'Extraindo texto' },
      { stage: 'tables', progress: 60, message: 'Extraindo tabelas' },
      { stage: 'ocr', progress: 75, message: 'OCR (se necessário)' },
      { stage: 'embedding', progress: 90, message: 'Gerando embeddings' },
    ];

    let i = 0;
    const ticker = setInterval(() => {
      if (i < stages.length) {
        eventId++;
        const payload = { type: 'progress', documentId: id, ...stages[i], timestamp: new Date().toISOString() };
        reply.raw.write(`id: ${eventId}\n`);
        reply.raw.write(`event: progress\n`);
        reply.raw.write(`data: ${JSON.stringify(payload)}\n\n`);
        i++;
      } else {
        eventId++;
        const done = { type: 'completed', documentId: id, timestamp: new Date().toISOString() };
        reply.raw.write(`id: ${eventId}\n`);
        reply.raw.write(`event: completed\n`);
        reply.raw.write(`data: ${JSON.stringify(done)}\n\n`);
        clearInterval(ticker);
      }
    }, 1000);

    const cleanup = () => { clearInterval(heartbeat); clearInterval(ticker); };
    request.raw.on('close', cleanup);
  }
);
```

CREATE EXTENSION IF NOT EXISTS pgcrypto;
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pg_trgm;

CREATE TABLE IF NOT EXISTS users (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  email TEXT UNIQUE NOT NULL,
  company TEXT NOT NULL,
  plan TEXT NOT NULL CHECK (plan IN ('free', 'pro', 'enterprise')),
  api_key_hash TEXT NOT NULL,
  api_secret_hash TEXT NOT NULL,
  credits_remaining INTEGER DEFAULT 1000,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW(),
  last_active TIMESTAMPTZ,
  metadata JSONB DEFAULT '{}'::jsonb
);

CREATE TABLE IF NOT EXISTS documents (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  tenant_id TEXT NOT NULL DEFAULT 'default',
  original_name TEXT NOT NULL,
  storage_key TEXT NOT NULL,
  size_bytes BIGINT NOT NULL,
  mime_type TEXT NOT NULL,
  sha256_hash TEXT NOT NULL,
  status TEXT NOT NULL DEFAULT 'pending',
  pages INTEGER,
  processing_time_ms INTEGER,
  error_message TEXT,
  case_number TEXT,
  court TEXT,
  parties JSONB,
  document_type TEXT,
  bates_prefix TEXT,
  bates_start INTEGER,
  bates_end INTEGER,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  processed_at TIMESTAMPTZ,
  expires_at TIMESTAMPTZ,
  metadata JSONB DEFAULT '{}'::jsonb,
  CONSTRAINT valid_status CHECK (status IN ('pending', 'processing', 'completed', 'failed'))
);

CREATE TABLE IF NOT EXISTS document_pages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  document_id UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
  page_number INTEGER NOT NULL,
  text TEXT,
  ocr_confidence FLOAT,
  word_count INTEGER,
  char_count INTEGER,
  checksum TEXT,
  bates_number TEXT,
  redactions JSONB DEFAULT '[]'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(document_id, page_number)
);

CREATE TABLE IF NOT EXISTS chunks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  document_id UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
  page_number INTEGER NOT NULL,
  chunk_index INTEGER NOT NULL,
  text TEXT NOT NULL,
  embedding vector(384),
  token_count INTEGER,
  metadata JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(document_id, page_number, chunk_index)
);

CREATE TABLE IF NOT EXISTS audit_log (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  document_id UUID REFERENCES documents(id) ON DELETE CASCADE,
  user_id UUID REFERENCES users(id) ON DELETE SET NULL,
  actor TEXT NOT NULL,
  action TEXT NOT NULL,
  ip_address INET,
  user_agent TEXT,
  payload JSONB DEFAULT '{}'::jsonb,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS webhooks (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
  url TEXT NOT NULL,
  events TEXT[] NOT NULL,
  secret TEXT NOT NULL,
  active BOOLEAN DEFAULT true,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS webhook_attempts (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  webhook_id UUID NOT NULL REFERENCES webhooks(id) ON DELETE CASCADE,
  event TEXT NOT NULL,
  payload JSONB NOT NULL,
  response_status INTEGER,
  response_body TEXT,
  error_message TEXT,
  attempt_number INTEGER DEFAULT 1,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS evidence_packages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  document_id UUID NOT NULL REFERENCES documents(id) ON DELETE CASCADE,
  manifest JSONB NOT NULL,
  files TEXT[] NOT NULL,
  package_hash TEXT NOT NULL,
  created_by UUID REFERENCES users(id),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE INDEX idx_documents_user_id ON documents(user_id);
CREATE INDEX idx_documents_status ON documents(status);
CREATE INDEX idx_documents_sha256 ON documents(sha256_hash);
CREATE INDEX idx_chunks_document_id ON chunks(document_id);
CREATE INDEX idx_chunks_embedding ON chunks USING ivfflat (embedding vector_cosine_ops);
CREATE INDEX idx_audit_log_document ON audit_log(document_id);
CREATE INDEX idx_audit_log_created ON audit_log(created_at);
CREATE INDEX idx_webhook_attempts_webhook ON webhook_attempts(webhook_id);

CREATE OR REPLACE FUNCTION update_updated_at()
RETURNS TRIGGER AS $$
BEGIN
  NEW.updated_at = NOW();
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER update_users_updated_at BEFORE UPDATE ON users
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

CREATE TRIGGER update_webhooks_updated_at BEFORE UPDATE ON webhooks
FOR EACH ROW EXECUTE FUNCTION update_updated_at();

1.5 CORREÇÕES DE SEGURANÇA E ERROR HANDLING
--------------------------------------------
Arquivo: paperflow-api/src/app.ts

Adicionar após registrar plugins:

```typescript
app.addHook('onRequest', async (req) => {
  if (req.headers['x-api-key']) {
    req.headers['x-api-key'] = '[REDACTED]';
  }
  if ((req.query as any)?.apiKey) {
    (req.query as any).apiKey = '[REDACTED]';
  }
});

app.setErrorHandler((err, req, reply) => {
  const traceId = crypto.randomUUID();
  const status = (err as any).statusCode || 500;
  
  req.log.error({
    err,
    traceId,
    url: req.url,
    method: req.method,
    ip: req.ip,
  });
  
  reply.status(status).send({
    error: {
      code: (err as any).code || 'INTERNAL_SERVER_ERROR',
      message: status < 500 ? err.message : 'An unexpected error occurred',
      traceId,
      ...(env.NODE_ENV !== 'production' && { stack: err.stack }),
    }
  });
});
```
